cmake_minimum_required(VERSION 3.20)
project(Electric3D VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(electric_core STATIC src/core.cpp src/simulation.cpp)
target_include_directories(electric_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(electric_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(electric_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  add_executable(Electric3D WIN32 src/main_win.cpp)
  target_link_libraries(Electric3D PRIVATE electric_core user32 gdi32 comdlg32 shell32)
  target_compile_definitions(Electric3D PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(Electric3D PRIVATE /W4 /utf-8)
  endif()
  set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT Electric3D)
endif()

enable_testing()
add_executable(electric_tests tests/test_core.cpp)
target_link_libraries(electric_tests PRIVATE electric_core)
target_compile_definitions(electric_tests PRIVATE ELECTRIC3D_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core_tests COMMAND electric_tests)
